# Keep test sources as .in files so AzerothCore's recursive source collector
# cannot accidentally link a test main() or mock headers into worldserver.
configure_file(CoreStubs.h.in CoreStubs.h COPYONLY)
foreach(header Chat CommandScript Config Log ObjectAccessor Player PlayerScript World WorldScript)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${header}.h" "#include \"CoreStubs.h\"\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/GlobalScript.h" "#include \"CoreStubs.h\"\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/LFG.h" "#include \"CoreStubs.h\"\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/LFGMgr.h" "#include \"CoreStubs.h\"\n")
configure_file(progression_tests.cpp.in progression_tests.cpp COPYONLY)
add_executable(progression_tests "${CMAKE_CURRENT_BINARY_DIR}/progression_tests.cpp")
target_compile_features(progression_tests PRIVATE cxx_std_17)
target_include_directories(progression_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../src")
add_test(NAME progression_behavior COMMAND progression_tests)

file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/raid-stubs")
configure_file(RaidResetStubs.h.in raid-stubs/RaidResetStubs.h COPYONLY)
foreach(header Config DBCStores GameTime InstanceSaveMgr Log World WorldScript)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/raid-stubs/${header}.h" "#include \"RaidResetStubs.h\"\n")
endforeach()
configure_file(raid_reset_tests.cpp.in raid-stubs/raid_reset_tests.cpp COPYONLY)
add_executable(raid_reset_tests "${CMAKE_CURRENT_BINARY_DIR}/raid-stubs/raid_reset_tests.cpp")
target_compile_features(raid_reset_tests PRIVATE cxx_std_17)
target_include_directories(raid_reset_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/raid-stubs" "${CMAKE_CURRENT_SOURCE_DIR}/../src")
add_test(NAME raid_reset_behavior COMMAND raid_reset_tests)

# Optional integration test executes the selected core's actual startup scheduler.
# CI supplies each supported patched core. Database and event storage are doubles.
if(PROGRESSION_CORE_SOURCE)
    file(READ "${PROGRESSION_CORE_SOURCE}/src/server/game/Instances/InstanceSaveMgr.cpp" reset_source)
    string(FIND "${reset_source}" "void InstanceSaveMgr::LoadResetTimes()" reset_start)
    string(FIND "${reset_source}" "void InstanceSaveMgr::LoadInstanceSaves()" reset_end)
    if(reset_start LESS 0 OR reset_end LESS reset_start)
        message(FATAL_ERROR "Cannot locate core reset loader for integration test")
    endif()
    math(EXPR reset_length "${reset_end} - ${reset_start}")
    string(SUBSTRING "${reset_source}" ${reset_start} ${reset_length} reset_loader)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/raid-stubs/core-load-reset-times.inc" "${reset_loader}")
    string(FIND "${reset_source}" "void InstanceSaveMgr::Update()" update_start)
    string(FIND "${reset_source}" "void InstanceSaveMgr::_ResetSave(" update_end)
    if(update_start LESS 0 OR update_end LESS update_start)
        message(FATAL_ERROR "Cannot locate core reset queue for integration test")
    endif()
    math(EXPR update_length "${update_end} - ${update_start}")
    string(SUBSTRING "${reset_source}" ${update_start} ${update_length} reset_update)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/raid-stubs/core-reset-update.inc" "${reset_update}")
    configure_file(core_reset_tests.cpp.in raid-stubs/core_reset_tests.cpp COPYONLY)
    add_executable(core_reset_tests "${CMAKE_CURRENT_BINARY_DIR}/raid-stubs/core_reset_tests.cpp")
    target_compile_features(core_reset_tests PRIVATE cxx_std_17)
    target_include_directories(core_reset_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/raid-stubs" "${CMAKE_CURRENT_SOURCE_DIR}/../src")
    add_test(NAME core_reset_migration COMMAND core_reset_tests)
    set_tests_properties(core_reset_migration PROPERTIES TIMEOUT 10)
endif()
